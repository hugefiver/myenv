# 这是 Hyprland 的主体配置，按键单独放在 keybinds.conf，方便你直接改。

$mod = SUPER
$terminal = wezterm
$fileManager = dolphin
$menu = rofi -show drun -show-icons

exec-once = ~/.config/hypr/autostart.sh

env = XCURSOR_SIZE,24
env = HYPRCURSOR_SIZE,24
env = NIXOS_OZONE_WL,1
env = XDG_CURRENT_DESKTOP,Hyprland
env = XDG_SESSION_TYPE,wayland

# Hyprland 不支持 Qt text-input-v2（仅 KWin 支持），
# 必须通过 fcitx IM module 为 Qt/GTK 应用提供输入法。
env = QT_IM_MODULE,fcitx
env = QT_IM_MODULES,wayland;fcitx
env = GTK_IM_MODULE,fcitx
env = SDL_IM_MODULE,fcitx

input {
	kb_layout = us
	kb_options = ctrl:swapcaps
	follow_mouse = 1
	repeat_rate = 45
	repeat_delay = 280

	touchpad {
		natural_scroll = yes
		tap-to-click = yes
		scroll_factor = 0.9
	}
}

general {
	gaps_in = 5
	gaps_out = 12
	border_size = 2
	layout = dwindle
	resize_on_border = true
	allow_tearing = false
	col.active_border = rgba(7dc4e4ff) rgba(b4befeef) rgba(cba6f7ff) 45deg
	col.inactive_border = rgba(313244cc)
}

decoration {
	rounding = 18
	active_opacity = 0.96
	inactive_opacity = 0.91
	fullscreen_opacity = 1.0
	dim_inactive = true
	dim_strength = 0.08

	shadow {
		enabled = true
		range = 30
		render_power = 3
		color = rgba(00000055)
		color_inactive = rgba(00000030)
	}

	blur {
		enabled = true
		size = 6
		passes = 4
		new_optimizations = true
		xray = true
		noise = 0.015
		contrast = 0.9
		brightness = 0.85
		vibrancy = 0.15
	}
}

animations {
	enabled = yes

	# Material Design 3 风格贝塞尔曲线
	bezier = md3_decel, 0.05, 0.7, 0.1, 1
	bezier = md3_accel, 0.3, 0, 0.8, 0.15
	bezier = overshot, 0.05, 0.9, 0.1, 1.05
	bezier = smoothOut, 0.36, 0, 0.66, -0.56
	bezier = smoothIn, 0.25, 1, 0.5, 1

	animation = windows, 1, 5, md3_decel, popin 60%
	animation = windowsIn, 1, 5, md3_decel, popin 60%
	animation = windowsOut, 1, 4, md3_accel, popin 60%
	animation = windowsMove, 1, 4, md3_decel
	animation = border, 1, 10, default
	animation = borderangle, 1, 120, smoothIn, loop
	animation = fade, 1, 4, md3_decel
	animation = workspaces, 1, 5, md3_decel, slide
	animation = specialWorkspace, 1, 4, md3_decel, slidefadevert -20%

	preserve_split = true
	smart_split = true
}

	new_status = master
}

	disable_splash_rendering = true
	focus_on_activate = true
	mouse_move_enables_dpms = true
	key_press_enables_dpms = true
	vfr = true
}

bindm = $mod, mouse:272, movewindow
bindm = $mod, mouse:273, resizewindow

# ── Layer Rules ──────────────────────────────────────────────
# waybar / rofi / wlogout 等 layer shell 表面的毛玻璃效果
layerrule = blur, waybar
layerrule = ignorealpha 0.3, waybar
layerrule = blur, rofi
layerrule = ignorealpha 0.3, rofi
layerrule = blur, logout_dialog
layerrule = ignorealpha 0.3, logout_dialog

bindel = , XF86MonBrightnessUp, exec, brightnessctl s 10%+
bindel = , XF86MonBrightnessDown, exec, brightnessctl s 10%-
bindel = , XF86AudioRaiseVolume, exec, wpctl set-volume -l 1.5 @DEFAULT_AUDIO_SINK@ 5%+
bindel = , XF86AudioLowerVolume, exec, wpctl set-volume @DEFAULT_AUDIO_SINK@ 5%-
bindel = , XF86AudioMute, exec, wpctl set-mute @DEFAULT_AUDIO_SINK@ toggle
bindl = , XF86AudioPlay, exec, playerctl play-pause
bindl = , XF86AudioNext, exec, playerctl next
bindl = , XF86AudioPrev, exec, playerctl previous
